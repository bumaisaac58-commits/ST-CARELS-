// Safe app bootstrap for ST CARELS. // This file does not assume external Supabase scripts are loaded. (function () { const app = window.stCarelsApp || {}; app.isSupabaseReady = function () { return !!(window.supabase && typeof window.supabase.createClient === 'function'); }; app.getSupabaseClient = function () { if (!this.isSupabaseReady()) return null; return window.supabase; }; app.showMessage = function (message, type = 'info') { const box = document.getElementById('app-message') || document.createElement('div'); if (!box.id) { box.id = 'app-message'; box.style.padding = '12px'; box.style.margin = '10px 0'; box.style.borderRadius = '8px'; document.body.prepend(box); } box.textContent = message; box.style.backgroundColor = type === 'error' ? '#fee2e2' : type === 'success' ? '#dcfce7' : '#dbeafe'; box.style.color = type === 'error' ? '#991b1b' : type === 'success' ? '#166534' : '#1e40af'; }; window.stCarelsApp = app; })(); console.log('St Carels app utilities loaded.');
